    package main

    import (

    func newRootCmd() *cobra.Command {
        app := &cli.App{
            Stdout: os.Stdout,
            Stderr: os.Stderr,
        }

        cmd := &cobra.Command{
            Use:          "tempus",
            Short:        "A multilingual ICS calendar file generator",
            SilenceUsage: true,
            PersistentPreRunE: cli.SetupPersistentPreRunE(app),
        }

        cmd.PersistentFlags().StringP("language", "l", "", "Language (en, es, pt, ga)")
        cmd.PersistentFlags().StringP("timezone", "t", "", "Timezone (IANA)")
        cmd.PersistentFlags().StringP("config", "c", "", "Config file path")

        cmd.AddCommand(
            cli.NewCreateCmd(app),
            cli.NewQuickCmd(app),
            cli.NewBatchCmd(app),
            cli.NewLintCmd(app),
            cli.NewConfigCmd(app),
            cli.NewInitCmd(app),
            cli.NewVersionCmd(app, version, commit, date),
            cli.NewTemplateCmd(app),
            cli.NewTimezoneCmd(app),
            cli.NewRRuleHelperCmd(app),
            cli.NewLocaleCmd(app),
        )

        return cmd
    }
