    ```go
    package cli

    func runInteractive(app *App, cmd *cobra.Command) error {
        // Variables for all wizard fields
        var (
            summary     string
            startDate   string
            startTime   string
            duration    string
            allDay      bool
            timezone    string
            alarmProf   string
            customAlarm string
            categories  []string
            customCat   string
            location    string
            description string
            confirmed   bool
        )

        // Pre-fill from config (per D-01, D-02)
        timezone = app.Config.Timezone
        alarmProf = app.Config.DefaultAlarmProfile
        if alarmProf == "" {
            alarmProf = "adhd-default"
        }

        // Build the 7-step form
        form := huh.NewForm(
            // Step 1/7 — Summary (per D-01)
            huh.NewGroup(
                huh.NewInput().
                    Title("Event name").
                    Value(&summary).
                    Validate(func(s string) error {
                        if strings.TrimSpace(s) == "" {
                            return fmt.Errorf("event name is required")
                        }
                        return nil
                    }),
            ).Title("Step 1/7 - Summary"),

            // Step 2/7 — Date, Time & Duration (per D-01)
            huh.NewGroup(
                huh.NewInput().Title("Start date (YYYY-MM-DD)").Value(&startDate).
                    Validate(func(s string) error {
                        if strings.TrimSpace(s) == "" {
                            return fmt.Errorf("start date is required")
                        }
                        return nil
                    }),
                huh.NewInput().Title("Start time (HH:MM, leave empty for all-day)").Value(&startTime),
                huh.NewInput().Title("Duration (e.g. 1h, 1h30m, 45m)").Value(&duration),
                huh.NewConfirm().Title("All-day event?").Value(&allDay),
            ).Title("Step 2/7 - Date, Time & Duration"),

            // Step 3/7 — Timezone (per D-01)
            huh.NewGroup(
                huh.NewInput().
                    Title("Timezone (IANA, e.g. America/New_York)").
                    Value(&timezone).
                    Validate(func(s string) error {
                        if strings.TrimSpace(s) == "" {
                            return fmt.Errorf("timezone is required")
                        }
                        return config.ValidateTimezone(s)
                    }),
            ).Title("Step 3/7 - Timezone"),

            // Step 4/7 — Alarms (per D-02)
            huh.NewGroup(
                huh.NewSelect[string]().
                    Title("Alarm profile").
                    Options(
                        huh.NewOption("ADHD Default (-2h, -30m, -5m)", "adhd-default"),
                        huh.NewOption("ADHD Countdown (-1h, -45m, -30m, -15m, -5m)", "adhd-countdown"),
                        huh.NewOption("Medication (-30m, -5m)", "medication"),
                        huh.NewOption("Single (-15m)", "single"),
                        huh.NewOption("None", "none"),
                        huh.NewOption("Custom...", "custom"),
                    ).
                    Value(&alarmProf),
            ).Title("Step 4/7 - Alarms"),

            // Step 4b — Custom alarm follow-up (per D-02: shown only when "custom" selected)
            huh.NewGroup(
                huh.NewInput().
                    Title("Custom alarm offsets (comma-separated, e.g. -2h,-30m,-5m)").
                    Value(&customAlarm).
                    Validate(func(s string) error {
                        if strings.TrimSpace(s) == "" {
                            return fmt.Errorf("at least one alarm offset is required")
                        }
                        return nil
                    }),
            ).Title("Step 4/7 - Custom Alarms").WithHideFunc(func() bool {
                return alarmProf != "custom"
            }),

            // Step 5/7 — Categories (per D-03)
            huh.NewGroup(
                huh.NewMultiSelect[string]().
                    Title("Categories (optional, use space to select)").
                    Options(
                        huh.NewOption("Work", "work"),
                        huh.NewOption("Health", "health"),
                        huh.NewOption("Personal", "personal"),
                        huh.NewOption("Travel", "travel"),
                        huh.NewOption("School", "school"),
                        huh.NewOption("Finance", "finance"),
                        huh.NewOption("Other...", "other"),
                    ).
                    Value(&categories),
            ).Title("Step 5/7 - Categories"),

            // Step 5b — Custom category follow-up (per D-03: "Other" entry)
            huh.NewGroup(
                huh.NewInput().
                    Title("Custom category name").
                    Value(&customCat),
            ).Title("Step 5/7 - Custom Category").WithHideFunc(func() bool {
                for _, c := range categories {
                    if c == "other" { return false }
                }
                return true
            }),

            // Step 6/7 — Details (per D-01)
            huh.NewGroup(
                huh.NewInput().Title("Location (optional)").Value(&location),
                huh.NewInput().Title("Description (optional)").Value(&description),
            ).Title("Step 6/7 - Details"),

            // Step 7/7 — Confirm (per D-01)
            // Note: huh.Note for summary display, then Confirm
            huh.NewGroup(
                huh.NewConfirm().
                    Title("Create this event?").
                    Affirmative("Yes, create it").
                    Negative("No, cancel").
                    Value(&confirmed),
            ).Title("Step 7/7 - Confirm"),
        )

        if err := form.Run(); err != nil {
            // Ctrl+C or terminal error — clean exit
            return nil
        }

        if !confirmed {
            fmt.Fprintf(app.Stdout, "Event creation cancelled.\n")
            return nil
        }

        // Post-process categories: replace "other" with customCat
        processedCategories := processCategories(categories, customCat)

        // Determine alarm specs
        alarmSpecs := resolveAlarmSpecs(alarmProf, customAlarm)

        // Determine if all-day: also infer if startTime is empty
        if startTime == "" {
            allDay = true
        }

        // Build createOptions from wizard values
        outputDir := app.Config.OutputDir
        if outputDir == "" {
            outputDir = "."
        }
        output := filepath.Join(outputDir, Slugify(summary)+".ics")

        opts := &createOptions{
            summary:     summary,
            startStr:    buildStartStr(startDate, startTime),
            durStr:      duration,
            startTZ:     timezone,
            allDay:      allDay,
            categories:  processedCategories,
            alarms:      alarmSpecs,
            location:    location,
            description: description,
            output:      output,
            prepTime:    false,
        }

        // Parse times using unified parsing
        parseOpts := parsing.ParseOptions{
            StartDate: startDate,
            StartTime: startTime,
            Duration:  duration,
            Timezone:  timezone,
            AllDay:    allDay,
            Summary:   summary,
        }
        result, err := parsing.Parse(parseOpts)
        if err != nil {
            return fmt.Errorf("invalid date/time: %w", err)
        }

        // Create calendar with event
        cal, err := createCalendarWithEvent(opts, result.Start, result.End)
        if err != nil {
            return err
        }

        // Write output
        if err := writeCalendarOutput(cal, opts.output); err != nil {
            return err
        }

        PrintOK(app.Stdout, "> Created: %s\n", opts.output)
        return nil
    }

    // processCategories replaces "other" sentinel with the custom category value
    func processCategories(categories []string, customCat string) []string { ... }

    // resolveAlarmSpecs converts profile name or custom string to alarm spec slice
    func resolveAlarmSpecs(profile, custom string) []string { ... }

    // buildStartStr combines date and time into a single start string
    func buildStartStr(date, time string) string { ... }
    ```

    **Key implementation details:**
    - Use `huh.WithHideFunc` for conditional groups (Step 4b custom alarm, Step 5b custom category) per RESEARCH.md Open Questions 1 and 2
    - Step 7 confirm: Before the confirm form group, print a summary table to `app.Stdout` showing all values. Use `huh.NewNote().Title("Event Summary").Description(summaryText)` inside the confirm group to display values inline.
    - Build the summary description string from all wizard values: summary, date, time, timezone, alarm profile, categories, location, description
    - `resolveAlarmSpecs`: if profile is "custom", split customAlarm by comma; if profile is "none", return empty; otherwise return `[]string{"profile:" + profile}`
    - `processCategories`: filter out "other" from slice, append customCat if non-empty
